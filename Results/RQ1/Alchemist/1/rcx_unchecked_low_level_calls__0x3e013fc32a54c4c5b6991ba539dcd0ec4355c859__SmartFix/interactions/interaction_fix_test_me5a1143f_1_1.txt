import { expect } from "chai";
import { ethers } from "hardhat";

describe("MultiplicatorX4 reference (ethers v6; deploy may require constructor arguments)", function () {
  it("should kill mutant me5a1143f by calling Command with a specific ether amount and verifying the recipient receives exactly that amount, not msg.value+1", async function () {
    const [owner, addr1, addr2] = await ethers.getSigners();
    const Factory = await ethers.getContractFactory("MultiplicatorX4");
    const instance = await Factory.deploy();
    await instance.waitForDeployment();

    // Fund the contract with some initial balance for the multiplicate function check
    await owner.sendTransaction({
      to: await instance.getAddress(),
      value: ethers.parseEther("1")
    });

    // Get initial balance of recipient
    const recipientInitialBalance = await ethers.provider.getBalance(addr2.address);

    // Call Command with 2 ether - original would send 2 ether, mutant would try to send 2 ether + 1 wei
    const txAmount = ethers.parseEther("2");
    const data = "0x";
    
    // The mutant will try to send txAmount + 1 wei which exceeds the msg.value, causing a revert
    await expect(
      instance.connect(owner).Command(addr2.address, data, { value: txAmount })
    ).to.be.reverted;

    // Verify recipient balance remained unchanged (transaction reverted)
    const recipientFinalBalance = await ethers.provider.getBalance(addr2.address);
    expect(recipientFinalBalance).to.equal(recipientInitialBalance);
  });
});
