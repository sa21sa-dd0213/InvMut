import { expect } from "chai";
import { ethers } from "hardhat";

describe("Roulette mutant mc4c6acff detection test", function () {
  it("should kill the mutant by sending 10 ether at a block where block.number % 15 == 0 and verifying the balance transfer", async function () {
    const [owner, player] = await ethers.getSigners();
    const Factory = await ethers.getContractFactory("Roulette");
    const instance = await Factory.deploy();
    await instance.waitForDeployment();

    // Fund the contract with initial balance so it has something to transfer
    await owner.sendTransaction({
      to: await instance.getAddress(),
      value: ethers.parseEther("10")
    });

    // Find a block where block.number % 15 == 0
    let targetBlock;
    const currentBlock = await ethers.provider.getBlockNumber();
    const remainder = currentBlock % 15;
    if (remainder === 0) {
      targetBlock = currentBlock;
    } else {
      targetBlock = currentBlock + (15 - remainder);
    }

    // Mine blocks until we reach the target block
    while ((await ethers.provider.getBlockNumber()) < targetBlock) {
      await ethers.provider.send("evm_mine", []);
    }

    // Get player's balance before the call
    const playerBalanceBefore = await ethers.provider.getBalance(player.address);
    const contractBalanceBefore = await ethers.provider.getBalance(await instance.getAddress());

    // Player sends exactly 10 ether to trigger the fallback
    const tx = await player.sendTransaction({
      to: await instance.getAddress(),
      value: ethers.parseEther("10")
    });
    await tx.wait();

    // Check that the player received the contract balance (original behavior)
    const playerBalanceAfter = await ethers.provider.getBalance(player.address);
    const expectedTransfer = contractBalanceBefore; // Original transfers entire balance
    const actualTransfer = playerBalanceAfter - playerBalanceBefore;
    
    // In the original, player receives contract balance (minus gas costs approximately)
    // In the mutant, condition is false, so player does NOT receive the balance
    expect(actualTransfer).to.be.gt(ethers.parseEther("9")); // Should receive ~10+ ether
  });
});
