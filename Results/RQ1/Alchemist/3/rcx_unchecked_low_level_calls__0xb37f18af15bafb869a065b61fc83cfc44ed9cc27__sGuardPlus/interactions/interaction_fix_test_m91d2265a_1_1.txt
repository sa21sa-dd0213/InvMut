import { expect } from "chai";
import { ethers } from "hardhat";

describe("SimpleWallet", function () {
  let simpleWallet: any;
  let owner: any;
  let addr1: any;
  let addr2: any;

  beforeEach(async function () {
    [owner, addr1, addr2] = await ethers.getSigners();
    
    const SimpleWallet = await ethers.getContractFactory("SimpleWallet");
    simpleWallet = await SimpleWallet.deploy();
    await simpleWallet.deployed();
  });

  describe("Deployment", function () {
    it("Should set the right owner", async function () {
      expect(await simpleWallet.owner()).to.equal(owner.address);
    });

    it("Should start with depositsCount = 0", async function () {
      expect(await simpleWallet.depositsCount()).to.equal(0);
    });
  });

  describe("Transactions", function () {
    it("Should accept ether and increase depositsCount", async function () {
      const tx = {
        to: simpleWallet.address,
        value: ethers.utils.parseEther("1.0"),
      };
      
      await owner.sendTransaction(tx);
      
      expect(await simpleWallet.depositsCount()).to.equal(1);
      expect(await ethers.provider.getBalance(simpleWallet.address)).to.equal(
        ethers.utils.parseEther("1.0")
      );
    });

    it("Should withdraw all ether correctly", async function () {
      // Send ether to contract first
      await owner.sendTransaction({
        to: simpleWallet.address,
        value: ethers.utils.parseEther("2.0"),
      });

      const initialBalance = await ethers.provider.getBalance(owner.address);
      
      const tx = await simpleWallet.withdrawAll();
      const receipt = await tx.wait();
      const gasCost = receipt.gasUsed.mul(receipt.effectiveGasPrice);
      
      const finalBalance = await ethers.provider.getBalance(owner.address);
      
      expect(finalBalance).to.equal(
        initialBalance.sub(gasCost).add(ethers.utils.parseEther("2.0"))
      );
      expect(await ethers.provider.getBalance(simpleWallet.address)).to.equal(0);
    });

    it("Should allow owner to send ether to another address", async function () {
      // Fund the contract

        to: simpleWallet.address,
        value: ethers.utils.parseEther("3.0"),

      await simpleWallet.sendMoney(
        addr1.address,

      expect(finalAddr1Balance).to.equal(
        initialAddr1Balance.add(ethers.utils.parseEther("1.5"))
      );
    });

      await expect(

    });
