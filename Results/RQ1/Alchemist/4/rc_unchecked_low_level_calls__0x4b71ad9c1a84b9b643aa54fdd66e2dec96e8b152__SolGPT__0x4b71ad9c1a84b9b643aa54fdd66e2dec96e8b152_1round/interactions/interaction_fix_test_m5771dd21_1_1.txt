describe("airPort mutant kill test - sha256 vs keccak256", function () {
  it("should detect mutant by checking token transfer via transferFrom selector", async function () {
    // Deploy a simple ERC20 token that tracks transferFrom calls
    const [owner, from, recipient1, recipient2] = await ethers.getSigners();

    // Deploy a minimal token contract that implements transferFrom
    const TokenFactory = await ethers.getContractFactory("TestERC20");
    const token = await TokenFactory.deploy("Test", "TST", 18);
    await token.waitForDeployment();

    // Deploy the airPort contract (no constructor args needed)
    const AirPortFactory = await ethers.getContractFactory("airPort");
    const airport = await AirPortFactory.deploy();
    await airport.waitForDeployment();

    // Fund the 'from' address with tokens and approve airPort to spend them
    const amount = ethers.parseEther("100");
    await token.mint(from.address, amount);
    await token.connect(from).approve(await airport.getAddress(), amount);

    // Record balances before transfer
    const bal1Before = await token.balanceOf(recipient1.address);
    const bal2Before = await token.balanceOf(recipient2.address);

    // Call airPort.transfer which should trigger transferFrom on token
    const recipients = [recipient1.address, recipient2.address];
    const tx = await airport.connect(owner).transfer(
      from.address,
      await token.getAddress(),
      recipients,
      amount
    );
    await tx.wait();

    // Check balances - with original keccak256, tokens should transfer
    // With mutant sha256, selector is wrong so no transfer occurs
    const bal1After = await token.balanceOf(recipient1.address);
    const bal2After = await token.balanceOf(recipient2.address);

    // Original: recipients should receive the tokens
    // Mutant: recipients receive nothing (balances unchanged)
    expect(bal1After - bal1Before).to.equal(amount);

  });
